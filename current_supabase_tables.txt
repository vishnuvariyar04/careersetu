CREATE TABLE public.companies (
  company_id uuid NOT NULL,
  name text NOT NULL,
  description text,
  industry text,
  created_at timestamp with time zone DEFAULT now(),
  mission text,
  vision text,
  employee_count integer,
  evaluation_metrics text[] DEFAULT '{}',
  policies jsonb,
  onboarding_advanced jsonb,
  CONSTRAINT companies_pkey PRIMARY KEY (company_id),
  CONSTRAINT companies_company_id_fkey FOREIGN KEY (company_id) REFERENCES auth.users(id)
);
CREATE TABLE public.environment_participants (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  environment_id uuid UNIQUE,
  student_id uuid,
  joined_at timestamp with time zone DEFAULT now(),
  CONSTRAINT environment_participants_pkey PRIMARY KEY (id),
  CONSTRAINT environment_participants_environment_id_fkey FOREIGN KEY (environment_id) REFERENCES public.virtual_environments(environment_id),
  CONSTRAINT environment_participants_student_id_fkey FOREIGN KEY (student_id) REFERENCES public.students(student_id)
);
-- NOTE: UNIQUE(environment_id) allows only one row per environment globally (likely unintended).
-- Prefer UNIQUE(environment_id, student_id) so many students can join the same environment.
-- App: POST /api/student/join-environment inserts here after the student has companies_joined.

CREATE TABLE public.job_openings (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  company_id uuid NOT NULL,
  title text NOT NULL,
  description text,
  required_skills text[] DEFAULT '{}',
  min_experience_level text DEFAULT 'beginner'::text CHECK (
    min_experience_level = ANY (ARRAY['beginner'::text, 'intermediate'::text, 'advanced'::text])
  ),
  job_type text DEFAULT 'internship'::text CHECK (
    job_type = ANY (ARRAY['internship'::text, 'full_time'::text, 'part_time'::text, 'contract'::text])
  ),
  location text,
  status text DEFAULT 'open'::text CHECK (
    status = ANY (ARRAY['open'::text, 'closed'::text, 'filled'::text])
  ),
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT job_openings_pkey PRIMARY KEY (id),
  CONSTRAINT job_openings_company_id_fkey FOREIGN KEY (company_id) REFERENCES public.companies(company_id)
);
CREATE TABLE public.job_applications (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  job_id uuid NOT NULL,
  student_id uuid NOT NULL,
  status text DEFAULT 'matched'::text CHECK (
    status = ANY (ARRAY['matched'::text, 'applied'::text, 'shortlisted'::text, 'interviewed'::text, 'offered'::text, 'rejected'::text])
  ),
  match_score integer,
  match_reasons jsonb DEFAULT '[]'::jsonb,
  cover_letter text,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  CONSTRAINT job_applications_pkey PRIMARY KEY (id),
  CONSTRAINT job_applications_job_id_fkey FOREIGN KEY (job_id) REFERENCES public.job_openings(id),
  CONSTRAINT job_applications_student_id_fkey FOREIGN KEY (student_id) REFERENCES public.students(student_id),
  CONSTRAINT job_applications_unique UNIQUE (job_id, student_id)
);

CREATE TABLE public.experience (
  experience_id uuid NOT NULL DEFAULT gen_random_uuid(),
  student_id uuid NOT NULL,
  company_name text NOT NULL,
  role text NOT NULL,
  exp_years interval,  -- Using INTERVAL to store 'time' durations accurately
  technologies_used text[],  -- Using an array for easier filtering of tech stacks

  created_at timestamp with time zone DEFAULT now(),
